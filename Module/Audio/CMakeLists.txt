cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(OpenglLearnAudio LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 20)
    enable_testing()
endif()
add_library(audio_module STATIC Private/audio_module.cpp Private/native_output.cpp)
target_include_directories(audio_module PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_compile_features(audio_module PUBLIC cxx_std_20)
if(WIN32)
    target_link_libraries(audio_module PRIVATE winmm)
elseif(APPLE)
    target_link_libraries(audio_module PRIVATE "-framework AudioToolbox")
else()
    find_package(Threads REQUIRED)
    target_link_libraries(audio_module PRIVATE Threads::Threads)
endif()
add_executable(audio_module_test Test/audio_module_test.cpp)
target_link_libraries(audio_module_test PRIVATE audio_module)
add_test(NAME audio_module_test COMMAND audio_module_test)
