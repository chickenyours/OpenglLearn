set(LUA_ROOT ${CMAKE_SOURCE_DIR}/third_party/lua-5.4.9/src)
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS ${LUA_ROOT}/*.c)
list(FILTER LUA_SOURCES EXCLUDE REGEX "/(lua|luac)\\.c$")
if(NOT LUA_SOURCES)
    message(FATAL_ERROR "Vendored Lua 5.4.9 source is missing")
endif()
add_library(iwanna_lua STATIC ${LUA_SOURCES})
target_include_directories(iwanna_lua PUBLIC ${LUA_ROOT})
if(UNIX)
    target_link_libraries(iwanna_lua PUBLIC m)
endif()
add_library(scripting_module STATIC Private/lua_module.cpp)
target_include_directories(scripting_module PUBLIC ${CMAKE_SOURCE_DIR}/Module)
target_link_libraries(scripting_module PUBLIC iwanna_lua)
